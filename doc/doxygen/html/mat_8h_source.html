128  Matrix<double,SX,SY> Mat::toMatrix() const {
129  Matrix<double,SX,SY> ret;
130  if (msize!=SX || nsize!=SY) {
131  if (msize==1 && SY==1 && nsize==SX) { // 允许广义向量切换
132  for (quint32 i=0;i<SX;++i)
133  ret(i,0) = operator()(0,i);
134  return ret;
135  }
136  if (nsize==1 && SX==1 && msize==SY) { // 允许
137  for (quint32 i=0;i<SY;++i)
138  ret(0,i) = operator()(i,0);
139  return ret;
140  }
141  throw MatExcept("Matrix size does not match %1,%2, is %3,%4 instead.",SX,SY,msize,nsize);
142  }
143  for (quint32 i=0;i<SX;++i)
144  for (quint32 j=0;j<SY;++j)
145  ret(i,j) = operator()(i,j);
146  return ret;
147  }
