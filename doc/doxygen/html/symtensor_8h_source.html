297 inline void SymTensor::adjustTrace(const double &newTrace) {
298  static const double d1d3 = 0.33333333333333333333333333333333;
299  /* --- 应变差 --- */
300  double dx = s11() - s22();
301  double dy = s22() - s33();
302  double dz = s33() - s11();
303  rs11() = (newTrace + dx - dz) * d1d3;
304  rs22() = (newTrace + dy - dx) * d1d3;
305  rs33() = (newTrace + dz - dy) * d1d3;
