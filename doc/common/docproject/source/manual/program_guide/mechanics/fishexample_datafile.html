; 文件名：ballexample.dat
;
;  展示 ball.pos FISH 内置函数的基本用法
; =============================================================================
model new

fish define get_ball_position_vector(bid)
  ; 获取 ID 为 bid 的球颗粒的位置矢量
  local b = ball.find(bid) ; 获取 ID 为 bid 的球颗粒地址
  local v = ball.pos(b)   ; 读取地址为 b 的球颗粒位置
  get_ball_position_vector = v
end

fish define set_ball_position_component(bid,idir,val)
  ; 获取 ID 为 bid 的球颗粒位置矢量的 idir 分量
  local b = ball.find(bid) ; 获取 ID 为 bid 的球颗粒地址
  ball.pos(b,idir) = val ; 写入位置的 idir 分量
end

; 设置模型计算域并创建球颗粒
model domain extent -1 1
ball create id 1 position-x 0.0 position-y 0.0 radius 0.1

; 创建全局 FISH 变量 vp1，执行 get_ball_position_vector 
; 函数，并将程序返回值存入 vp1
[global vp1 = get_ball_position_vector(1)]
fish list [vp1]  ; 在控制台窗格中输出 vp1 的值

; 执行 set_ball_position_component 函数，赋予新的 x 分量
[set_ball_position_component(1,1,0.1)]

; 执行 set_ball_position_component 函数，赋予新的 y 分量
[set_ball_position_component(1,2,0.2)]

; 利用内嵌 FISH 创建全局 FISH 变量 vp2，执行 
; get_ball_position_vector 函数，并将程序返回值存入 vp2
[vp2 = get_ball_position_vector(1)]
fish list [vp2]  ; 在控制台窗格中输出 vp2 的值

program return
